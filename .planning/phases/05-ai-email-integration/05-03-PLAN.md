**use-faq.ts** -- Create new hooks file:

```typescript
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { api } from '@/lib/api';

export interface Faq {
  id: string;
  question: string;
  answer: string;
  tags: string[];
  createdAt: string;
  updatedAt: string;
}

export function useFaqs(tag?: string) {
  return useQuery({
    queryKey: ['faqs', { tag }],
    queryFn: async () => {
      const params = tag ? { tag } : undefined;
      const response = await api.get<{ data: Faq[] }>('/api/v1/faqs', { params });
      return response.data;
    },
  });
}

export function useCreateFaq() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: (data: { question: string; answer: string; tags?: string[] }) =>
      api.post<{ data: Faq }>('/api/v1/faqs', data).then(r => r.data),
    onSuccess: () => qc.invalidateQueries({ queryKey: ['faqs'] }),
  });
}

export function useUpdateFaq() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: ({ id, ...data }: { id: string; question?: string; answer?: string; tags?: string[] }) =>
      api.patch<{ data: Faq }>(`/api/v1/faqs/${id}`, data).then(r => r.data),
    onSuccess: () => qc.invalidateQueries({ queryKey: ['faqs'] }),
  });
}

export function useDeleteFaq() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: (id: string) => api.delete(`/api/v1/faqs/${id}`),
    onSuccess: () => qc.invalidateQueries({ queryKey: ['faqs'] }),
  });
}
```
