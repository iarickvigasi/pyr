    ```typescript
    /** WebSocket RPC frame types for OpenClaw Gateway protocol v3 */

    // -- Outbound frames (backend -> gateway) --

    export interface RpcRequest {
      type: 'req';
      id: string;
      method: string;
      params?: unknown;
    }

    export interface ConnectParams {
      minProtocol: number;
      maxProtocol: number;
      client: {
        id: string;
        version: string;
        platform: string;
        mode: string;
      };
      role: string;
      scopes: string[];
      auth: { token: string };
    }

    export interface ChatSendParams {
      sessionKey: string;
      message: string;
      idempotencyKey: string;
    }

    export interface AgentParams {
      message: string;
      agentId: string;
      sessionKey: string;
      deliver: boolean;
      idempotencyKey: string;
      extraSystemPrompt?: string;
    }

    export interface SessionsResetParams {
      key: string;
      reason: string;
    }

    // -- Inbound frames (gateway -> backend) --

    export interface RpcResponse {
      type: 'res';
      id: string;
      ok: boolean;
      payload?: unknown;
      error?: { code: string; message: string };
    }

    export interface GatewayEvent {
      type: 'event';
      event: string;
      payload?: unknown;
    }

    export type InboundFrame = RpcResponse | GatewayEvent;

    // -- Chat event payload --

    export interface ChatEvent {
      runId: string;
      sessionKey: string;
      seq: number;
      state: 'delta' | 'final' | 'error' | 'aborted';
      message?: unknown;
      usage?: {
        prompt_tokens: number;
        completion_tokens: number;
        cache_read_input_tokens?: number;
        cache_creation_input_tokens?: number;
      };
      model?: string;
      errorMessage?: string;
    }

    // -- HelloOk payload from connect response --

    export interface HelloOkPayload {
      type: 'hello-ok';
      protocol: number;
      policy?: {
        tickIntervalMs?: number;
      };
    }

    // -- GatewayWsClient options --

    export interface GatewayWsClientOptions {
      url: string;
      token: string;
      reconnectMs?: number;
      requestTimeoutMs?: number;
    }
    ```

    1. Create `packages/backend/src/services/gateway/gateway-ws-client.ts`:

    Build a `GatewayWsClient` class extending `EventEmitter` that:
    - **Constructor** accepts `GatewayWsClientOptions` and a pino logger.
    - **`start()`** method: Creates a `ws` WebSocket connection to `opts.url`. Handles the connect handshake:
      - Waits for `connect.challenge` event from gateway
      - Sends `connect` RPC request with `{ minProtocol: 3, maxProtocol: 3, client: { id: 'pyr-backend', version: '1.0.0', platform: 'linux', mode: 'backend' }, role: 'operator', scopes: ['operator.read', 'operator.write'], auth: { token } }`
      - Waits for `hello-ok` response, extracts `tickIntervalMs` from policy
      - Sets `connected = true`
      - Starts tick timeout monitoring: if no message arrives within `2 * tickIntervalMs`, trigger reconnect
      - On WebSocket `close` or `error`, if not intentionally closed, trigger reconnect
    - **Reconnection logic**: exponential backoff starting at `reconnectMs` (default 3000), capped at 30000ms, with jitter. Rejects all pending requests on disconnect. Auto-reconnects unless `stop()` was called.
    - **`request<T>(method, params?)`**: Sends an RPC request frame (`{ type: 'req', id: crypto.randomUUID(), method, params }`) and returns a Promise that resolves when the matching `res` frame arrives. Timeout after `requestTimeoutMs` (default 120000ms). Stores pending requests in a `Map<string, { resolve, reject, timer }>`.
    - **`onChatEvent(handler)`**: Registers a listener for `chat` events. Returns an unsubscribe function. The handler receives a typed `ChatEvent`. Multiple listeners can be active simultaneously (for concurrent dashboard chat + draft generation).
    - **`stop()`**: Sets `closed = true`, clears reconnect timer, rejects all pending requests with "Client closing", closes the WebSocket.
    - **`get isConnected()`**: Returns current connection state.
    - **Message routing** (in the WebSocket `message` handler):
      - Parse JSON. Reset tick timeout on any received message.
      - If `frame.type === 'res'`: find pending request by `frame.id`, resolve or reject based on `frame.ok`.
      - If `frame.type === 'event' && frame.event === 'chat'`: emit to all registered chat event listeners with `frame.payload` typed as `ChatEvent`.
      - If `frame.type === 'event' && frame.event === 'shutdown'`: log warning, prepare for reconnect.
      - If `frame.type === 'event' && frame.event === 'connect.challenge'`: handle handshake (only during initial connect).

    Use `import WebSocket from 'ws'` (default import), `import crypto from 'crypto'`, and `import { EventEmitter } from 'events'`.

    - Calls `gateway.start()` but does NOT await it (non-blocking startup -- gateway might not be ready when backend starts). Log "Gateway WebSocket connection initiated" at info level. Catch and log start errors.
    - Decorates `fastify.gateway = client`.
    - Registers `onClose` hook that calls `gateway.stop()`.

    - Use `ws` library (not native WebSocket) for ping/pong support and custom headers
    - The client must handle the case where `start()` is called but the gateway is not yet available -- it should retry connection via the reconnect logic, not throw
    - All pending request rejections on disconnect should use a descriptive error: "Gateway connection lost"
    - Log reconnection attempts with backoff delay at warn level
    - Log successful connections at info level
    - Use pino logger passed from Fastify (not console)
